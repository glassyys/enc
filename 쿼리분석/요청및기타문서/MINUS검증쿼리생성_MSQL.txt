MINUS 비교: 전체 컬럼 사용
최종 SELECT 조회: col02, col03 제외

SET SESSION group_concat_max_len = 100000;

WITH META AS (
    SELECT tablename, col_idx, col_name, data_type, part_col_yn
    FROM db_schma.table_metadata
    WHERE tablename = 'tbl01'
),
COL_INFO AS (
    SELECT
          tablename
        , GROUP_CONCAT(col_name ORDER BY col_idx SEPARATOR ', ') AS col_list
        , GROUP_CONCAT(CASE WHEN TRIM(part_col_yn) = 'Y' THEN col_name END ORDER BY col_idx SEPARATOR ', ') AS pk_list
        , GROUP_CONCAT(CASE WHEN TRIM(part_col_yn) = 'Y' THEN CONCAT('A.', col_name, ' <=> B.', col_name) END ORDER BY col_idx SEPARATOR ' AND ') AS pk_join_dev
        , GROUP_CONCAT(CASE WHEN TRIM(part_col_yn) = 'Y' THEN CONCAT('A.', col_name, ' <=> C.', col_name) END ORDER BY col_idx SEPARATOR ' AND ') AS pk_join_d2
        , GROUP_CONCAT(CONCAT('A.', col_name, ' AS DIFF_', col_name) ORDER BY col_idx SEPARATOR ', ') AS diff_cols
        , GROUP_CONCAT(CONCAT('B.', col_name, ' AS DEV_', col_name) ORDER BY col_idx SEPARATOR ', ') AS dev_cols
        , GROUP_CONCAT(CONCAT('C.', col_name, ' AS D2_', col_name) ORDER BY col_idx SEPARATOR ', ') AS d2_cols
    FROM META
    GROUP BY tablename
)
SELECT CONCAT(
    'WITH DEV_D2_DIFF AS (', CHAR(10),
    '    SELECT ', col_list, ', ''DEV_MINUS_D2'' AS DIFF_SIDE', CHAR(10),
    '    FROM midp_crs_dev.', tablename, CHAR(10),
    '    MINUS', CHAR(10),
    '    SELECT ', col_list, ', ''DEV_MINUS_D2'' AS DIFF_SIDE', CHAR(10),
    '    FROM midp_crs_d2.', tablename, CHAR(10),
    '),', CHAR(10),
    'D2_DEV_DIFF AS (', CHAR(10),
    '    SELECT ', col_list, ', ''D2_MINUS_DEV'' AS DIFF_SIDE', CHAR(10),
    '    FROM midp_crs_d2.', tablename, CHAR(10),
    '    MINUS', CHAR(10),
    '    SELECT ', col_list, ', ''D2_MINUS_DEV'' AS DIFF_SIDE', CHAR(10),
    '    FROM midp_crs_dev.', tablename, CHAR(10),
    '),', CHAR(10),
    'ALL_DIFF AS (', CHAR(10),
    '    SELECT ', col_list, ', DIFF_SIDE FROM midp_crs_dev_D2_DIFF', CHAR(10),
    '    UNION ALL', CHAR(10),
    '    SELECT ', col_list, ', DIFF_SIDE FROM midp_crs_d2_DEV_DIFF', CHAR(10),
    ')', CHAR(10),
    'SELECT', CHAR(10),
    '       CASE', CHAR(10),
    '           WHEN A.DIFF_SIDE = ''DEV_MINUS_D2'' AND C.', SUBSTRING_INDEX(pk_list, ', ', 1), ' IS NULL THEN ''DEV_ONLY''', CHAR(10),
    '           WHEN A.DIFF_SIDE = ''D2_MINUS_DEV'' AND B.', SUBSTRING_INDEX(pk_list, ', ', 1), ' IS NULL THEN ''D2_ONLY''', CHAR(10),
    '           ELSE ''DATA_DIFF''', CHAR(10),
    '       END AS DIFF_TYPE,', CHAR(10),
    '       A.DIFF_SIDE,', CHAR(10),
    '       ', diff_cols, ',', CHAR(10),
    '       ', dev_cols, ',', CHAR(10),
    '       ', d2_cols, CHAR(10),
    'FROM ALL_DIFF A', CHAR(10),
    'LEFT JOIN dev.', tablename, ' B', CHAR(10),
    '  ON ', pk_join_dev, CHAR(10),
    'LEFT JOIN d2.', tablename, ' C', CHAR(10),
    '  ON ', pk_join_d2, ';'
) AS VALID_SQL
FROM COL_INFO
WHERE pk_list IS NOT NULL;

SET SESSION group_concat_max_len = 1000000;

SET @table_name      = 'tbl01';
SET @dev_schema      = 'midp_crs_dev';
SET @d2_schema       = 'midp_crs_d2';
SET @partition_value = '20261031';
SET @exclude_cols    = 'col02,col03';   -- 비교 제외 컬럼 (콤마 구분)

WITH MD AS (
    SELECT
        col_idx,
        col_name,
        (COALESCE(part_col_yn, 'N') = 'Y') AS is_part
    FROM db_schma.table_metadata
    WHERE tablename = @table_name
),
POS AS (
    SELECT MAX(CASE WHEN is_part = 1 THEN col_idx END) AS max_part_idx
    FROM MD
),
NXT AS (
    SELECT MIN(MD.col_idx) AS next_idx
    FROM MD
    CROSS JOIN POS
    WHERE MD.col_idx > POS.max_part_idx
),
COLS AS (
    SELECT
        MD.col_idx,
        MD.col_name,
        MD.is_part,
        /* 키 컬럼 = 파티션 컬럼 + 그 다음 첫 컬럼(next_col) */
        CASE WHEN MD.is_part = 1 OR MD.col_idx = NXT.next_idx
             THEN 1 ELSE 0 END AS is_key,
        /* 비교 컬럼 = 키 컬럼 + 제외 목록에 없는 컬럼 */
        CASE WHEN MD.is_part = 1
                OR MD.col_idx = NXT.next_idx
                OR FIND_IN_SET(
                       LOWER(MD.col_name),
                       LOWER(REPLACE(@exclude_cols, ' ', ''))
                   ) = 0
             THEN 1 ELSE 0 END AS is_cmp
    FROM MD
    CROSS JOIN NXT
),
META AS (
    SELECT
        GROUP_CONCAT(CASE WHEN is_cmp = 1 THEN col_name END
                     ORDER BY col_idx SEPARATOR ', ') AS cmp_cols,
        GROUP_CONCAT(CASE WHEN is_cmp = 1 THEN CONCAT('A.', col_name) END
                     ORDER BY col_idx SEPARATOR ', ') AS a_cols,
        GROUP_CONCAT(CASE WHEN is_key = 1 THEN col_name END
                     ORDER BY col_idx SEPARATOR ', ') AS key_cols,
        GROUP_CONCAT(CASE WHEN is_key = 1
                          THEN CONCAT('A.', col_name, ' <=> X.', col_name) END
                     ORDER BY col_idx SEPARATOR ' AND ') AS key_join,
        GROUP_CONCAT(CASE WHEN is_part = 1
                          THEN CONCAT(col_name, ' = ''', @partition_value, '''') END
                     ORDER BY col_idx SEPARATOR ' AND ') AS partition_where
    FROM COLS
)
SELECT
    CONCAT(
        'WITH D2_BASE AS (', CHAR(10),
        '    SELECT DISTINCT ', M.cmp_cols, CHAR(10),
        '    FROM ', @d2_schema, '.', @table_name, CHAR(10),
        '    WHERE ', M.partition_where, CHAR(10),
        '), DEV_BASE AS (', CHAR(10),
        '    SELECT DISTINCT ', M.cmp_cols, CHAR(10),
        '    FROM ', @dev_schema, '.', @table_name, CHAR(10),
        '    WHERE ', M.partition_where, CHAR(10),
        '), D2_DIFF AS (', CHAR(10),
        '    SELECT ', M.cmp_cols, ' FROM D2_BASE', CHAR(10),
        '    MINUS', CHAR(10),
        '    SELECT ', M.cmp_cols, ' FROM DEV_BASE', CHAR(10),
        '), DEV_DIFF AS (', CHAR(10),
        '    SELECT ', M.cmp_cols, ' FROM DEV_BASE', CHAR(10),
        '    MINUS', CHAR(10),
        '    SELECT ', M.cmp_cols, ' FROM D2_BASE', CHAR(10),
        '), D2_KEY AS (', CHAR(10),
        '    SELECT DISTINCT ', M.key_cols, ', 1 AS HIT FROM D2_BASE', CHAR(10),
        '), DEV_KEY AS (', CHAR(10),
        '    SELECT DISTINCT ', M.key_cols, ', 1 AS HIT FROM DEV_BASE', CHAR(10),
        '), RESULT AS (', CHAR(10),
        '    SELECT', CHAR(10),
        '        CASE WHEN X.HIT IS NULL THEN ''D2_ONLY'' ELSE ''DATA_DIFF'' END AS DIFF_TYPE,', CHAR(10),
        '        ''D2'' AS DATA_SIDE,', CHAR(10),
        '        ', M.a_cols, CHAR(10),
        '    FROM D2_DIFF A', CHAR(10),
        '    LEFT JOIN DEV_KEY X ON ', M.key_join, CHAR(10),
        '    UNION ALL', CHAR(10),
        '    SELECT', CHAR(10),
        '        CASE WHEN X.HIT IS NULL THEN ''DEV_ONLY'' ELSE ''DATA_DIFF'' END AS DIFF_TYPE,', CHAR(10),
        '        ''DEV'' AS DATA_SIDE,', CHAR(10),
        '        ', M.a_cols, CHAR(10),
        '    FROM DEV_DIFF A', CHAR(10),
        '    LEFT JOIN D2_KEY X ON ', M.key_join, CHAR(10),
        ')', CHAR(10),
        'SELECT * FROM RESULT', CHAR(10),
        'ORDER BY ', M.key_cols, ', DATA_SIDE;'
    ) AS GENERATED_SPARK_SQL
FROM META M;

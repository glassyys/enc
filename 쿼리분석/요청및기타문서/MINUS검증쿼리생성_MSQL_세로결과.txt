SET @table_name = 'tbl01';
SET @dev_schema = 'midp_crs_dev';
SET @d2_schema = 'midp_crs_d2';
SET @partition_value = '20261031';

WITH META AS
(
    SELECT
        /* MINUS 비교 컬럼: col02, col03 제외 */
        GROUP_CONCAT(
            CASE
                WHEN part_col_yn = 'Y'
                  OR col_name NOT IN ('col02', 'col03')
                THEN col_name
            END
            ORDER BY col_idx
            SEPARATOR ', '
        ) AS minus_cols,

        /* 파티션 WHERE 조건 */
        GROUP_CONCAT(
            CASE
                WHEN part_col_yn = 'Y'
                THEN CONCAT(
                    col_name,
                    ' = ''',
                    @partition_value,
                    ''''
                )
            END
            ORDER BY col_idx
            SEPARATOR ' AND '
        ) AS partition_where,

        /* PK 출력 컬럼 */
        GROUP_CONCAT(
            CASE
                WHEN part_col_yn = 'Y'
                THEN CONCAT(
                    'A.', col_name,
                    ' AS PK_', col_name
                )
            END
            ORDER BY col_idx
            SEPARATOR ', '
        ) AS pk_output,

        /* 최종 조회 컬럼: 전체 컬럼 */
        GROUP_CONCAT(
            CONCAT(
                'B.', col_name, ' AS DEV_', col_name,
                ', C.', col_name, ' AS D2_', col_name
            )
            ORDER BY col_idx
            SEPARATOR ', '
        ) AS result_cols,

        /* DEV PK JOIN */
        GROUP_CONCAT(
            CASE
                WHEN part_col_yn = 'Y'
                THEN CONCAT(
                    'A.', col_name,
                    ' = B.', col_name
                )
            END
            ORDER BY col_idx
            SEPARATOR ' AND '
        ) AS pk_join_dev,

        /* D2 PK JOIN */
        GROUP_CONCAT(
            CASE
                WHEN part_col_yn = 'Y'
                THEN CONCAT(
                    'A.', col_name,
                    ' = C.', col_name
                )
            END
            ORDER BY col_idx
            SEPARATOR ' AND '
        ) AS pk_join_d2,

        /* 첫 번째 PK */
        SUBSTRING_INDEX(
            GROUP_CONCAT(
                CASE
                    WHEN part_col_yn = 'Y'
                    THEN col_name
                END
                ORDER BY col_idx
                SEPARATOR ', '
            ),
            ', ',
            1
        ) AS first_pk,

        /* PK 다음 순서의 컬럼 */
        (
            SELECT col_name
            FROM db_schma.table_metadata
            WHERE tablename = @table_name
              AND col_idx >
                  (
                      SELECT MAX(col_idx)
                      FROM db_schma.table_metadata
                      WHERE tablename = @table_name
                        AND part_col_yn = 'Y'
                  )
            ORDER BY col_idx
            LIMIT 1
        ) AS next_col

    FROM db_schma.table_metadata
    WHERE tablename = @table_name
),

/* PK 정렬 순서 생성 */
PK_SEQ AS
(
    SELECT
        col_idx,
        ROW_NUMBER() OVER (
            ORDER BY col_idx
        ) AS RN
    FROM db_schma.table_metadata
    WHERE tablename = @table_name
      AND part_col_yn = 'Y'
),

PK_ORDER AS
(
    SELECT
        GROUP_CONCAT(
            2 + RN
            ORDER BY col_idx
            SEPARATOR ', '
        ) AS order_cols
    FROM PK_SEQ
)

/* 최종 Spark SQL 생성 및 출력 */

        '    WHERE ', M.partition_where, CHAR(10),
        '    MINUS', CHAR(10),
        '    SELECT ', M.minus_cols,
        ' FROM ', @dev_schema, '.', @table_name, CHAR(10),
        '    WHERE ', M.partition_where, CHAR(10),
        '    LIMIT 3', CHAR(10),

        '), ALL_DIFF AS (', CHAR(10),
        '    SELECT ', M.minus_cols,
        ', ''DEV_MINUS_D2'' AS DIFF_SIDE FROM DEV_D2_DIFF',
        CHAR(10),
        '    UNION ALL', CHAR(10),
        '    SELECT ', M.minus_cols,
        ', ''D2_MINUS_DEV'' AS DIFF_SIDE FROM D2_DEV_DIFF',
        CHAR(10),
        ')', CHAR(10),

        'SELECT', CHAR(10),
        '    CASE', CHAR(10),
        '        WHEN A.DIFF_SIDE = ''DEV_MINUS_D2'' ',
        'AND C.', M.first_pk,
        ' IS NULL THEN ''DEV_ONLY''', CHAR(10),
        '        WHEN A.DIFF_SIDE = ''D2_MINUS_DEV'' ',
        'AND B.', M.first_pk,
        ' IS NULL THEN ''D2_ONLY''', CHAR(10),
        '        ELSE ''DATA_DIFF''', CHAR(10),
        '    END AS DIFF_TYPE,', CHAR(10),

        '    A.DIFF_SIDE,', CHAR(10),
        '    ', M.pk_output, ',', CHAR(10),
        '    ', M.result_cols, CHAR(10),

        'FROM ALL_DIFF A', CHAR(10),

        'LEFT JOIN ', @dev_schema, '.', @table_name,
        ' B ON ', M.pk_join_dev,
        ' AND A.', M.next_col, ' = B.', M.next_col,
        CHAR(10),

        'LEFT JOIN ', @d2_schema, '.', @table_name,
        ' C ON ', M.pk_join_d2,
        ' AND A.', M.next_col, ' = C.', M.next_col,
        CHAR(10),

        'ORDER BY 1, 2, ', P.order_cols, ';'

    ) AS GENERATED_SPARK_SQL

FROM META M
CROSS JOIN PK_ORDER P;
